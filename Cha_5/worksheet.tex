\title{Determinants}\author{T.T.}\date{}

\begin{chaptermap}\noindent The determinant is a single number with an unusual amount of information about a matrix. It detects invertibility, records the effect of elimination, and eventually produces formulas for inverses and solutions. Its linearity applies to one row at a time: the tempting rule $\det(A+B)=\det A+\det B$ is false.\par\smallskip{\footnotesize\raggedleft{}Adapted from Gilbert Strang, Section 5.1, pp.~247--248.\par}\end{chaptermap}
\topic{\initial{D}EFINITION AND \initial{P}ROPERTIES}
\[\det I=1,\qquad \det(AB)=\det(A)\det(B),\qquad A\text{ invertible}\Longleftrightarrow\det A\ne0.\]
\textbf{Row rules:} swapping rows changes the sign; multiplying a row by $c$ multiplies the determinant by $c$; adding a multiple of one row to another leaves it unchanged.
\textbf{Linearity:} the determinant is linear in each row (or column) separately and alternating in its rows.
\[\det({A}^{T})=\det A,\qquad \det({A}^{-1})={\det A}^{-1},\qquad \det(cA)=c^n\det A.\]
\topic{\initial{C}OMPUTATIONAL \initial{F}ORMULAE}
\textbf{Triangular matrices:} $\det A=\prod_{i=1}^n a_{ii}$.
\[\det A=\sum_{j=1}^n a_{ij}C_{ij}=\sum_{i=1}^n a_{ij}C_{ij},\quad C_{ij}={(-1)}^{i+j}\det(A_{ij}).\]
\textbf{Cofactor formula:} $\displaystyle {A}^{-1}=\frac{1}{\det A}{\operatorname{Cof}(A)}^{T}$.
\[\det A=\sum_{\sigma\in S_n}\operatorname{sgn}(\sigma)\prod_{i=1}^n a_{i,\sigma(i)}.\]
\textbf{Cramer's rule:} if $Ax=b$ and $\det A\ne0$, $\displaystyle x_i=\frac{\det A_i(b)}{\det A}$.
\textbf{Vandermonde determinant:} $\displaystyle \det\!\left({[x_j^{i-1}]}_{i,j=1}^n\right)=\prod_{1\leq i<j\leq n}(x_j-x_i)$.
\topic{\initial{G}EOMETRY AND \initial{V}OLUME}
\[\operatorname{volume}\bigl(A({[0,1]}^n)\bigr)=|\det A|,\qquad \det A=0\Longleftrightarrow\text{columns lie in a lower-dimensional subspace}.\]

\topic{\initial{P}ROBLEM \initial{S}ET}

\question{1} Use row operations to compute
\[
\det\begin{bmatrix}
1&2&3\\
2&5&7\\
1&1&1
\end{bmatrix}.
\]
State the determinant rule used at each step and identify the product of the pivots.

\question{2} Let $A$ be an $n\times n$ matrix and let $c\in\mathbb R$.
\subquestion{i}{Use linearity in one row at a time to show that
\[
\det(cA)=c^n\det A.
\]}
\subquestion{ii}{If $A$ is $4\times4$ and $\det A=-2$, find $\det(2A)$, $\det({A}^{T}A)$, and $\det(A^{-1})$.}
\subquestion{iii}{Show by a $2\times2$ counterexample that the determinant is not linear as a function of the whole matrix: in general, $\det(A+B)\ne\det A+\det B$.}

\question{3} Let $A$ be an $n\times n$ matrix and let $\boldsymbol{1}={(1,\ldots,1)}^{T}$.
\subquestion{i}{Suppose the entries in every row of $A$ add to zero. Show that $A\boldsymbol{1}=0$, and hence that $\det A=0$.}
\subquestion{ii}{Suppose instead that the entries in every row add to one. Show that $\det(A-I)=0$.}
\subquestion{iii}{Does the condition in part (ii) imply that $\det A=1$? Give a counterexample and explain what the condition actually proves.}

\question{4} Let $A=\begin{bmatrix}a_1&\cdots&a_n\end{bmatrix}$ be an $n\times n$ matrix.
\subquestion{i}{Assume that the columns of $A$ are linearly dependent. Use linearity of the determinant in one column and the repeated-column rule to prove that $\det A=0$.}
\subquestion{ii}{Prove the converse using the connection between the determinant and invertibility.}
\subquestion{iii}{Explain why this equivalence agrees with the volume interpretation of $|\det A|$.}

\question{5} Let $Q$ be a real orthogonal matrix, so that ${Q}^{T}Q=I$.
\subquestion{i}{Prove that $\det Q=1$ or $\det Q=-1$.}
\subquestion{ii}{For a $2\times2$ orthogonal matrix, write its first column as ${ (\cos\theta,\sin\theta) }^{T}$. Show that its second column must be either ${ (-\sin\theta,\cos\theta) }^{T}$ or ${ (\sin\theta,-\cos\theta) }^{T}$, and identify the determinant in each case.}

\newpage
\question{6} Suppose that $A$ is an invertible matrix with integer entries.
\subquestion{i}{Prove that if $\det A=1$ or $\det A=-1$, then every entry of $A^{-1}$ is an integer.}
\subquestion{ii}{Prove the converse: if both $A$ and $A^{-1}$ have integer entries, then $\det A=1$ or $\det A=-1$.}

\question{7} Let $A=\begin{bmatrix}a_1&\cdots&a_n\end{bmatrix}$ be invertible, and let $A_j(b)$ denote the matrix obtained by replacing column $j$ of $A$ by $b$. Suppose that $Ax=b$.
\subquestion{i}{Write $b$ as a linear combination of the columns of $A$.}
\subquestion{ii}{Use linearity of the determinant in column $j$ to show that
\[
\det A_j(b)=x_j\det A.
\]
Explain why every other term in the expansion is zero.}
\subquestion{iii}{Deduce Cramer's rule.}

\question{8} Let $C=(C_{ij})$ be the cofactor matrix of an $n\times n$ matrix $A$.
\subquestion{i}{Show that every diagonal entry of $A{C}^{T}$ equals $\det A$.}
\subquestion{ii}{Show that every off-diagonal entry of $A{C}^{T}$ is zero. Hint: interpret it as a cofactor expansion of a matrix with two equal rows.}
\subquestion{iii}{Deduce the identity
\[
A{C}^{T}=(\det A)I,
\]
and hence derive the cofactor formula for $A^{-1}$ when $\det A\ne0$.}
