\begin{chaptermap}\noindent Elimination is a systematic way to solve linear equations. It removes one unknown at a time until the system is upper triangular; back substitution then works from the bottom upward. Each elimination step changes the equations without changing their common solution.\par\smallskip{\footnotesize\raggedleft{}Adapted from Gilbert Strang, Section 2.2, pp.~46--47.\par}\end{chaptermap}
\topic{\initial{S}YSTEMS AND \initial{E}LIMINATION}
\[Ax=b,\qquad Ax=x_1a_1+\cdots+x_n a_n.\]
\textbf{Row operations:} interchange rows; multiply a row by a nonzero constant; add a multiple of one row to another.
\textbf{Rank criterion:} $Ax=b$ is solvable iff $\operatorname{rank}(A)=\operatorname{rank}([A\ b])$.
\textbf{Pivot theorem:} free variables occur in non-pivot columns; if $r$ is the rank, the solution set has $n-r$ free parameters.
\[Ax=0\Rightarrow x=x_{\rm pivot}+x_{\rm free};\quad x=x_p+x_n,\quad Ax_n=0.\]
\textbf{Rouché--Capelli conclusion:} a consistent system has either one solution (no free variables) or infinitely many solutions.
\topic{\initial{M}ATRIX \initial{O}PERATIONS}
\[A^{-1}A=I,\quad {(AB)}^{-1}=B^{-1}A^{-1},\quad {(A^{T})}^{-1}={(A^{-1})}^{T}.\]
\[E_{ij}^{-1}=E_{ij}(-c),\quad {E_i(c)}^{-1}=E_i(c^{-1}),\quad P^{-1}=P^{T}.\]
\textbf{Elimination factorization:} $EA=U$, hence $A=LU$; with row exchanges, $PA=LU$.
\textbf{LU solution:} solve $Ly=b$ followed by $Ux=y$.
\topic{\initial{S}PECIAL \initial{M}ATRICES}
\[A^{T}A\text{ is symmetric},\quad A=S+K,\quad S=\frac{A+A^{T}}{2},\quad K=\frac{A-A^{T}}{2}.\]
\textbf{Orthogonal matrix:} $Q^{T}Q=I\Longleftrightarrow Q^{-1}=Q^{T}$; it preserves dot products and lengths.
\[\det(AB)=\det A\det B,\qquad A^{-1}=\frac{1}{\det A}{\operatorname{Cof}(A)}^{T}.\]

\topic{\initial{P}ROBLEM \initial{S}ET}
\question{1} Consider the system
\[
2x+3y=1,\qquad 10x+9y=11.
\]
\subquestion{i}{Use elimination to obtain an upper triangular system and identify its two pivots.}
\subquestion{ii}{Solve the system by back substitution and verify the solution as a linear combination of the columns of the coefficient matrix.}
\subquestion{iii}{Without repeating the full elimination, solve the system when the right side is changed to ${(4,44)}^{T}$.}

\question{2} Suppose that $Ax=b$ has two distinct solutions $x_1$ and $x_2$.
\subquestion{i}{Show that $(1-t)x_1+tx_2$ is also a solution for every $t\in\mathbb R$.}
\subquestion{ii}{Deduce that a system of linear equations cannot have exactly two solutions.}

\question{3} Let
\[
A=\begin{bmatrix}1&1&0\\4&6&1\\-2&2&0\end{bmatrix},
\qquad
b=\begin{bmatrix}1\\0\\0\end{bmatrix}.
\]
\subquestion{i}{Find the elimination matrices $E_{21}$, $E_{31}$, and $E_{32}$ that reduce $A$ to an upper triangular matrix $U$.}
\subquestion{ii}{Compute $M=E_{32}E_{31}E_{21}$ and verify that $MA=U$.}
\subquestion{iii}{Apply the same elimination steps to $b$ and solve $Ax=b$ by back substitution.}

\question{4} Let
\[
A=\begin{bmatrix}1&2\\0&0\end{bmatrix},
\qquad
B=\begin{bmatrix}1&0\\3&0\end{bmatrix}.
\]
\subquestion{i}{Compute $AB$ and $BA$, and show that matrix multiplication is not commutative in this example.}
\subquestion{ii}{Expand ${(A+B)}^{2}$ without assuming that $A$ and $B$ commute.}
\subquestion{iii}{Compute ${(A+B)}^{2}$ and $A^2+2AB+B^2$. Explain precisely when the scalar-style identity ${(A+B)}^{2}=A^2+2AB+B^2$ is valid for square matrices.}

\question{5} Let $A$ be a $3\times3$ matrix.
\subquestion{i}{Suppose that the third row of $A$ is the sum of its first two rows. Show that $A$ is not invertible. Derive a necessary condition on $b={(b_1,b_2,b_3)}^{T}$ for $Ax=b$ to be consistent, and describe what happens to the third equation during elimination.}
\subquestion{ii}{Suppose instead that the third column of $A$ is the sum of its first two columns. Find a nonzero solution of $Ax=0$, and use it to show that $A$ is not invertible.}

\question{6} Use Gauss-Jordan elimination on $[A\ I]$ to find $A^{-1}$ for
\[
A=\begin{bmatrix}
2&1&0\\
1&2&1\\
0&1&2
\end{bmatrix}.
\]
Check your answer by matrix multiplication.

\question{7} Let
\[
A=\begin{bmatrix}
1&1&1\\
1&2&3\\
1&3&6
\end{bmatrix},
\qquad
b=\begin{bmatrix}5\\7\\11\end{bmatrix}.
\]
\subquestion{i}{Use elimination without row exchanges to factor $A=LU$, where $L$ has ones on its diagonal.}
\subquestion{ii}{Solve $Lc=b$ by forward substitution and then solve $Ux=c$ by back substitution.}

\question{8} Let $A\in\mathbb R^{m\times n}$ and $B\in\mathbb R^{n\times p}$.
\subquestion{i}{Prove from the entries of the matrices that ${(AB)}^{T}=B^{T}A^{T}$.}
\subquestion{ii}{Let $Q=[q_1\ \cdots\ q_n]$ be an $n\times n$ matrix satisfying $Q^{T}Q=I$. Show that the columns of $Q$ are unit vectors and are mutually orthogonal. Prove the converse.}
\subquestion{iii}{Show that $Q$ preserves dot products and lengths. Give a $2\times2$ example with first entry $q_{11}=\cos\theta$.}
